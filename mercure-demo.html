💻 Integration Quick Reference
// 1. JavaScript Frontend (Browser EventSource)
const eventSource = new EventSource('/.well-known/mercure?topic=' + encodeURIComponent('https://example.com/notifications'));
eventSource.onmessage = (e) => {
    console.log('Real-time message received:', e.data);
};

// 2. PHP Backend (Publishing via POST)
// Use the built-in helper in /var/www/mercure-test.php or post directly with your publisher JWT:
$response = file_get_contents('http://localhost/mercure-test.php', false, stream_context_create([
    'http' => [
        'method'  => 'POST',
        'header'  => 'Content-Type: application/x-www-form-urlencoded',
        'content' => http_build_query(['topic' => 'https://example.com/notifications', 'message' => 'Hello Realtime!'])
    ]
]));
